  netmask 255.255.255.252

auto eth1

  # Switch10
  address 192.213.10.1
  netmask 255.255.255.128

auto eth2

  # Deus-Protocol7
  address 192.213.10.129
  netmask 255.255.255.252

  up ip route add default via 192.213.14.1
  up ip route add 192.213.8.0/23 via 192.213.10.130
  up echo nameserver 192.168.122.1 > /etc/resolv.conf
